# Qlib pilot dependencies (Python 3.11)
# Pins keep the resolver away from known-breaking majors for pyqlib 0.9.6
# (numpy 2 / pandas 3 / mlflow 3 lines). joblib is pinned to 1.2.0 because
# qlib's ParallelExt (qlib/utils/paral.py) relies on joblib internals
# (_backend_args) that joblib >= 1.3 removed — with a newer joblib, qrun
# dies in dataset loading with AttributeError: 'ParallelExt' object has
# no attribute '_backend_args' (seen in run 42823639271, fixed by this pin).
# LightGBM arrives as a pyqlib dependency; it needs the OpenMP runtime
# (libgomp1), installed in the workflow's install step.

pandas==2.2.3
scipy==1.14.1
scikit-learn==1.5.2
joblib==1.2.0
mlflow>=2.0,<3
setuptools
